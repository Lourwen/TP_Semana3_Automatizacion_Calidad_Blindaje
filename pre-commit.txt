
#!/bin/sh

echo "===================================="
echo " PRE-COMMIT: AUDITORIA DE CALIDAD"
echo "===================================="

echo ""
echo "[1/3] Formatter: Spotless"
./gradlew.bat spotlessApply
if [ $? -ne 0 ]; then
    echo "ERROR: fallo Spotless."
    exit 1
fi

git add -u
if [ $? -ne 0 ]; then
    echo "ERROR: no se pudieron preparar los cambios de formato."
    exit 1
fi

echo ""
echo "[2/3] Linter: Checkstyle"
./gradlew.bat checkstyleMain checkstyleTest
if [ $? -ne 0 ]; then
    echo "ERROR: fallo Checkstyle."
    exit 1
fi

echo ""
echo "[3/3] Pruebas unitarias: JUnit"
./gradlew.bat test
if [ $? -ne 0 ]; then
    echo "ERROR: fallaron las pruebas unitarias."
    exit 1
fi

echo ""
echo "CALIDAD APROBADA: commit permitido."
exit 0
